// Rule 1: Lumma Stealer C2 API Communication (Critical)
// Detects HTTP communication with the Lumma Stealer C2 domain and API endpoint
// Deploy to: SIEM proxy logs, packet captures, network traffic analysis tools
rule CTI_Lumma_C2_SetAgent_Communication
{
    meta:
        description = "Detects Lumma Stealer C2 communication via /api/set_agent"
        author      = "Adebayo"
        family      = "LummaC2"
        mitre       = "T1071.001, T1041"
        severity    = "Critical"
    strings:
        $domain   = "sinitjq.cyou"   nocase
        $endpoint = "/api/set_agent"  nocase
        $actlog   = "act=log"         nocase
        $token    = "token="          nocase
    condition:
        ($domain and $endpoint and $token)
        or ($domain and $endpoint and $actlog)
}
